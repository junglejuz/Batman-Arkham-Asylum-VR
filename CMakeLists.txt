cmake_minimum_required(VERSION 3.21)
project(BatmanAsylumVR VERSION 0.1.5 LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Asylum requires MSVC Win32: -G Visual Studio 17 2022 -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_subdirectory(third_party)
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(batman_asylum SHARED ${CORE_SOURCES} src/hash.cpp src/game/adapter_registry.cpp
 src/game/asylum/adapter.cpp src/game/asylum/d3d9_bridge.cpp src/game/asylum/render_anchors.cpp
 src/game/asylum/native_stereo.cpp src/game/asylum/startup.cpp src/game/asylum/hud9.cpp src/game/asylum/eyesame9.cpp src/game/asylum/xform9.cpp src/game/asylum/fxstep9.cpp src/game/asylum/dlss.cpp
 src/game/asylum/window.cpp src/game/asylum/afw.cpp src/game/asylum/detail9.cpp src/game/asylum/pipeline_cache.cpp src/game/asylum/dynamic_pipeline.cpp src/game/asylum/managed9.cpp src/game/asylum/native9_bridge.cpp src/game/asylum/billboard9.cpp src/game/asylum/markers9.cpp src/game/asylum/menu_state.cpp src/game/asylum/transitions9.cpp src/game/asylum/first_person.cpp)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/bvr_version.h"
 "#pragma once\n#define BVR_VERSION \"${PROJECT_VERSION}\"\n#define BVR_BUILD_ID \"Asylum-0.1.5-PROFILECLEAN\"\n")
target_include_directories(batman_asylum PRIVATE src "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(batman_asylum PRIVATE minhook imgui openxr_loader d3d11 d3d12 dxgi d3dcompiler dbghelp windowscodecs ole32 bcrypt psapi dxguid)
target_compile_definitions(batman_asylum PRIVATE BVR_WITH_OPENXR)
add_library(d3d9 SHARED src/d3d9_proxy.cpp src/d3d9_proxy.def)
add_library(xinput1_3 SHARED src/proxy.cpp src/proxy.def)
add_executable(aa_xr_config32 tools/xr_config32.cpp)
target_link_libraries(aa_xr_config32 PRIVATE openxr_loader)
add_executable(asylum_offline_check tests/offline_check.cpp)
target_link_libraries(asylum_offline_check PRIVATE dbghelp)
add_executable(asylum_native9ex_check tests/native9ex_check.cpp src/game/asylum/managed9.cpp src/game/asylum/native9_bridge.cpp)
target_include_directories(asylum_native9ex_check PRIVATE src)
target_link_libraries(asylum_native9ex_check PRIVATE d3d11 dxgi d3dcompiler)
enable_testing()
add_test(NAME native9ex_resource_compatibility COMMAND asylum_native9ex_check)
add_test(NAME native9ex_assembled_mod COMMAND asylum_offline_check "$<TARGET_FILE:batman_asylum>" "$<TARGET_FILE:d3d9>" "$<TARGET_FILE:xinput1_3>" native)
add_executable(asylum_present_sequence_check tests/present_sequence_check.cpp)
target_include_directories(asylum_present_sequence_check PRIVATE src)
add_test(NAME native_present_sequence COMMAND asylum_present_sequence_check)
add_executable(asylum_opening_movies_check tests/opening_movies_check.cpp)
target_include_directories(asylum_opening_movies_check PRIVATE src)
add_test(NAME opening_movie_identity COMMAND asylum_opening_movies_check)
add_executable(asylum_stereo_projection_check tests/stereo_projection_check.cpp)
target_include_directories(asylum_stereo_projection_check PRIVATE src third_party/OpenXR-SDK/include)
add_test(NAME scene_and_canvas_projection COMMAND asylum_stereo_projection_check)
add_test(NAME render_size_fits_desktop COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/render_size_check.ps1")
add_test(NAME foreign_host_and_proxy_guard COMMAND asylum_offline_check "$<TARGET_FILE:batman_asylum>" "$<TARGET_FILE:d3d9>" "$<TARGET_FILE:xinput1_3>")
add_test(NAME d3d9_gpu_transport COMMAND asylum_offline_check "$<TARGET_FILE:batman_asylum>" "$<TARGET_FILE:d3d9>" "$<TARGET_FILE:xinput1_3>" gpu)
add_test(NAME overlay_gamepad_navigation COMMAND asylum_offline_check "$<TARGET_FILE:batman_asylum>" "$<TARGET_FILE:d3d9>" "$<TARGET_FILE:xinput1_3>" panel)
add_test(NAME graphics_profile_rollback COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/graphics_profile_check.ps1")
